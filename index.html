const WINNING_COMBINATIONS = [
  [0, 1, 2], [3, 4, 5], [6, 7, 8], // Rows
  [0, 3, 6], [1, 4, 7], [2, 5, 8], // Cols
  [0, 4, 8], [2, 4, 6]             // Diag
];

function checkWin() {
  return WINNING_COMBINATIONS.some(combo => {
    return combo.every(index => {
      return gameState[index] === currentPlayer;
    });
  });
}

The Minimax AI Decision Engine

The "Impossible" AI evaluates all possible future outcomes using a recursive Minimax decision tree algorithm.

+10 (AI Win) 0 (Tie) -10 (Human Win)

Minimax Logic Flow
function minimax(board, depth, isMaximizing) {
  let score = evaluateBoard(board);
  if (score === 10) return score - depth; // AI wins
  if (score === -10) return score + depth; // Player wins
  if (!hasMovesLeft(board)) return 0; // Tie

  if (isMaximizing) {
    let best = -Infinity;
    for (let cell of emptyCells(board)) {
      board[cell] = 'O'; // Make move
      best = Math.max(best, minimax(board, depth + 1, false));
      board[cell] = '';  // Undo move
    }
    return best;
  } else {
    let best = Infinity;
    for (let cell of emptyCells(board)) {
      board[cell] = 'X'; // Make move
      best = Math.min(best, minimax(board, depth + 1, true));
      board[cell] = '';  // Undo move
    }
    return best;
  }
}
